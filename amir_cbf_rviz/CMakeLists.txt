cmake_minimum_required(VERSION 3.8)
project(amir_cbf_rviz)
find_package(ament_cmake REQUIRED)
find_package(rviz_default_plugins REQUIRED)
find_package(rviz_common REQUIRED)
find_package(rviz_rendering REQUIRED)
find_package(pluginlib REQUIRED)
add_library(${PROJECT_NAME} SHARED src/foreground_marker_array.cpp)
target_compile_features(${PROJECT_NAME} PUBLIC cxx_std_14)
ament_target_dependencies(${PROJECT_NAME} rviz_default_plugins rviz_common rviz_rendering pluginlib)
pluginlib_export_plugin_description_file(rviz_common plugins.xml)
install(TARGETS ${PROJECT_NAME} LIBRARY DESTINATION lib)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_foreground test/test_foreground.cpp)
  ament_target_dependencies(test_foreground rviz_default_plugins rviz_common rviz_rendering pluginlib)
endif()
ament_package()
